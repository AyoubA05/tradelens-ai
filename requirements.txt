# Streamlit surface. Installed on Streamlit Community Cloud (Python 3.11).
# Shared runtime deps live in requirements-base.txt so the FastAPI image does
# not have to install Streamlit, PyArrow and Plotly — presentation dependencies
# with no business in a backend, which roughly triple the image.
-r requirements-base.txt
streamlit==1.50.0
# Streamlit 1.50's AppTest dataframe conversion segfaults natively with
# PyArrow 25 on Python 3.11. Version 21 is verified by the Journal interaction
# flows and prevents the CI process from exiting with signal 11.
pyarrow==21.0.0
plotly==6.7.0
# Streamlit 1.50 requires pillow<12, so this surface cannot take the 12.x line
# the API surface uses. It does not decode untrusted uploads — that path is
# FastAPI's — so the exposure differs from requirements-api.txt's.
pillow==11.3.0
