# FastAPI surface. Installed in the API and worker container, which does NOT
# install requirements.txt — Streamlit, PyArrow and Plotly are presentation
# dependencies with no business in a backend.
-r requirements-base.txt
# Pinned to versions that support BOTH Python 3.9 (local development) and 3.11
# (the container), so tests and production run identical code. uvicorn >=0.40
# and httpx >=0.29 require Python 3.10 and cannot be used until local moves up.
# Two lines per package, selected by interpreter.
#
# The container and CI run Python 3.11 and take the patched versions:
# FastAPI 0.120.x caps starlette below 0.50, and every starlette release that
# clears PYSEC-2026-161/248/249/2280/2281 needs 3.10+, so clearing the advisory
# means moving FastAPI too. Local development is still on 3.9 and keeps the old
# line; it is the only place that divergence exists, and CI is authoritative.
fastapi==0.120.4; python_version < "3.10"
fastapi==0.141.1; python_version >= "3.10"
# Pinned explicitly rather than left to FastAPI's open range, so the audited
# version is the deployed version.
starlette==1.6.0; python_version >= "3.10"
uvicorn[standard]==0.39.0; python_version < "3.10"
uvicorn[standard]==0.52.3; python_version >= "3.10"
# Cloudflare R2 is S3-compatible. Used by the storage adapter.
boto3==1.42.7
# Pillow 12.3.0 closes the July 2026 native-code/image-processing advisory set
# and is the production requirement for THIS surface, which is the one that
# decodes untrusted uploads. Pillow 12 dropped Python 3.9, so local development
# stays on 11.3.0; the container runs 3.11 and gets 12.3.0.
#
# It is pinned here rather than in requirements-base.txt because Streamlit 1.50
# requires pillow<12. A shared pin made the Streamlit surface unresolvable on
# Python 3.11 — pip cannot satisfy both — which is exactly how CI caught it.
pillow==11.3.0; python_version < "3.10"
pillow==12.3.0; python_version >= "3.10"
